function editarPagamento(id) { if (!ehAdmin) return; const pagamento = pagamentos.find(p => p.id === id); if (!pagamento) return; // Criar modal de edição const modalHtml = `

✏️ Editar Pagamento

`; // Adicionar modal ao body document.body.insertAdjacentHTML('beforeend', modalHtml); } function fecharModalEdicao() { const modal = document.getElementById('modalEdicao'); if (modal) modal.remove(); } function salvarEdicao(id) { const pagamento = pagamentos.find(p => p.id === id); if (!pagamento) return; const novaData = document.getElementById('editData').value; const novoValor = parseFloat(document.getElementById('editValor').value); const novaObs = document.getElementById('editObs').value; if (!novaData || isNaN(novoValor)) { alert('❌ Preencha todos os campos corretamente'); return; } // Registrar alterações const alteracoes = []; if (pagamento.data !== novaData) alteracoes.push(`data: ${pagamento.data} → ${novaData}`); if (pagamento.valor !== novoValor) alteracoes.push(`valor: R$ ${pagamento.valor} → R$ ${novoValor}`); if (pagamento.obs !== novaObs) alteracoes.push(`observação alterada`); // Atualizar pagamento pagamento.data = novaData; pagamento.valor = novoValor; pagamento.obs = novaObs; salvarDados(); fecharModalEdicao(); if (alteracoes.length > 0) { adicionarObservacaoAuto(`✏️ Pagamento #${id} editado por ${usuarioAtual}: ${alteracoes.join(', ')}`); } alert('✅ Pagamento atualizado com sucesso!'); }
